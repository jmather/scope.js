'use strict';

var fs = require('fs');
var path = require('path');
var _ = require('../internal/underscore');
var projectRoot = path.resolve(__dirname, '..', '..');

function fromRoot(modulePath) {
    return require(path.join(projectRoot, modulePath));
}

/**
 * @module DepBuilder
 */
module.exports = {
    loadPlugins: function(plugins) {
        var loaded = {};

        _.each(plugins, function(plugin) {
            var pluginPath = path.join(projectRoot, 'lib/plugins', plugin, 'index.js');
            if (fs.existsSync(pluginPath)) {
                loaded[plugin] = require(pluginPath);
            }
        });

        return loaded;
    },

    byFileName: function(filePath, plugins) {
        return this.byDirectory(filePath, plugins);
    },

    byDirectory: function(sourcePath, plugins) {
        var rootPath = path.dirname(sourcePath);
        var rootName = path.basename(sourcePath, '.test.js');
        var statePath = path.join(rootPath, rootName + '-state.json');
        var state = fs.existsSync(statePath) ? JSON.parse(fs.readFileSync(statePath, 'utf8')) : {};
        var configPath = path.join(rootPath, rootName + '-config.json');

        if (fs.existsSync(configPath) === false) {
            throw new Error('Could not find the config in ' + configPath);
        }

        var config = JSON.parse(fs.readFileSync(configPath, 'utf8'));
        return this.byPassThrough(state, config, plugins);
    },

    byPassThrough: function(state, config, plugins) {
        var DataManager = fromRoot('lib/vm/data-manager');
        var ValueManager = fromRoot('lib/vm/value-manager');
        var TypeManager = fromRoot('lib/vm/type-manager');
        var InputManager = fromRoot('lib/vm/input-manager');
        var QuestionManager = fromRoot('lib/vm/question-manager');
        var InstructionExecutor = fromRoot('lib/vm/plugins/scope/lib/instruction-executor');

        var dataManager = new DataManager(state);
        var typeManager = new TypeManager(plugins);
        var inputManager = new InputManager();
        var questionManager = new QuestionManager();
        var instructionExecutor = new InstructionExecutor(inputManager, questionManager, plugins);
        var valueManager = new ValueManager(config, dataManager, typeManager, instructionExecutor);

        return {
            dataManager: dataManager,
            typeManager: typeManager,
            inputManager: inputManager,
            questionManager: questionManager,
            instructionExecutor: instructionExecutor,
            valueManager: valueManager
        };
    }
};
