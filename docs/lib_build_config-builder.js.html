'use strict';

var fs = require('fs');
var path = require('path');
var _ = require('../internal/underscore');
var Transformer = require('../compiler/index');

/**
 * @typedef {Object} CompileOptions
 * @property {string} configPath
 * @property {string} outputPath
 * @property {string[]} pluginPaths
 * @property {boolean} pretty
 */


function existingDirectories(paths) {
    return paths.filter(function(candidate) {
        return fs.existsSync(candidate) && fs.statSync(candidate).isDirectory();
    });
}

function getTransformerModules(paths) {
    var transformers = [];

    _.each(existingDirectories(paths), function(pluginPath) {
        _.each(fs.readdirSync(pluginPath), function(file) {
            var transformationPath = path.join(pluginPath, file, 'transformation');

            if (fs.existsSync(transformationPath + '.js')) {
                transformers.push(require(transformationPath));
            }

            if (fs.existsSync(transformationPath + 's')) {
                _.each(fs.readdirSync(transformationPath + 's'), function(transformationFile) {
                    var candidate = path.join(transformationPath + 's', transformationFile);
                    if (fs.statSync(candidate).isFile()) {
                        transformers.push(require(candidate));
                    }
                });
            }
        });
    });

    return transformers;
}

function getPluginIndexes(paths) {
    var plugins = [];

    _.each(existingDirectories(paths), function(pluginPath) {
        _.each(fs.readdirSync(pluginPath), function(file) {
            var indexPath = path.join(pluginPath, file, 'index.js');
            if (fs.existsSync(indexPath)) {
                plugins.push({ name: file, path: path.resolve(indexPath) });
            }
        });
    });

    return plugins;
}

function pluginNames(plugins) {
    return plugins.map(function(plugin) {
        return 'scope-plugin-' + plugin.name;
    });
}

function requireSpecifier(fromDirectory, targetFile) {
    var relative = path.relative(fromDirectory, targetFile).split(path.sep).join('/');
    return relative.indexOf('.') === 0 ? relative : './' + relative;
}

function writePluginLoader(outputPath, plugins) {
    fs.mkdirSync(outputPath, { recursive: true });
    var entries = plugins.map(function(plugin) {
        return JSON.stringify(plugin.name) + ': require(' +
            JSON.stringify(requireSpecifier(outputPath, plugin.path)) + ')';
    });
    fs.writeFileSync(
        path.join(outputPath, 'plugins.js'),
        "'use strict';\nmodule.exports = {" + entries.join(',') + '};\n'
    );
}

/**
 *
 * @param {CompileOptions} options
 * @returns {{plugins: string[], config: Object}}
 */
function compile(options) {
    var pluginPaths = existingDirectories(options.pluginPaths || []);
    var transformerModules = getTransformerModules(pluginPaths);
    var transformers = _.map(transformerModules, function(TransformerClass) {
        return new TransformerClass({});
    });
    var transformer = new Transformer(transformers, ['init', 'copy', 'resolve', 'validate', 'metadata']);
    var data = Transformer.loadData([options.configPath]);
    var compiledData = transformer.transform(data);
    var plugins = getPluginIndexes(pluginPaths);

    compiledData.plugins = pluginNames(plugins);

    fs.mkdirSync(options.outputPath, { recursive: true });
    var json = JSON.stringify(compiledData, null, options.pretty ? 2 : 0);
    fs.writeFileSync(path.join(options.outputPath, 'grid-collection-config.json'), json + '\n');
    fs.writeFileSync(path.join(options.outputPath, 'config.js'), "'use strict';\nmodule.exports = " + json + ';\n');
    fs.writeFileSync(path.join(options.outputPath, 'config.json'), json + '\n');
    writePluginLoader(options.outputPath, plugins);

    return { config: compiledData, plugins: plugins };
}

module.exports = {
    compile: compile,
    getPluginIndexes: getPluginIndexes,
    getTransformerModules: getTransformerModules,
    writePluginLoader: writePluginLoader
};
